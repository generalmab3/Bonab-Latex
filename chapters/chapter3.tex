\section{نمای کلی}
چارچوب از چهار جزء ساخته می‌شود: داده بهره‌برداری، شبکه عصبی، باقیمانده‌های فیزیکی و بهینه‌ساز پارامتر. داده، مشاهده‌های بار، تولید، قیمت و وضعیت شارژ را می‌آورد. شبکه عصبی از این مشاهده‌ها تصمیم می‌سازد. تصمیم در معادلات توازن و باتری قرار می‌گیرد و باقیمانده محاسبه می‌شود. مجموع خطای داده و خطای فیزیکی پارامترها را به‌روز می‌کند.

\section{ورودی و خروجی}
بردار ورودی در زمان $t$ چنین گرفته می‌شود:
\begin{equation}
  \bm{x}_t=
  \left[t,\,P_{\mathrm{load}}(t),\,P_{\mathrm{PV}}(t),\,
  c_{\mathrm{buy}}(t),\,c_{\mathrm{sell}}(t),\,\mathrm{SOC}(t)\right].
  \label{eq:model-input}
\end{equation}
خروجی پیشنهادی عبارت است از
\begin{equation}
  \hat{\bm{y}}_t=
  \left[\hat{P}_{\mathrm{grid}}(t),\,
  \hat{P}_{\mathrm{ch}}(t),\,
  \hat{P}_{\mathrm{dis}}(t),\,
  \widehat{\mathrm{SOC}}(t+1)\right].
  \label{eq:model-output}
\end{equation}
اگر داده مرجع برای بعضی از این خروجی‌ها موجود نباشد، می‌توان آن مؤلفه را از $\mathcal{L}_{\mathrm{data}}$ حذف کرد و فقط با فیزیک مقید نمود. این انعطاف یکی از دلایل انتخاب \lr{PINN} است.

برای جلوگیری از شارژ و دشارژ هم‌زمان می‌توان خروجی را بازپارامتری کرد؛ مثلاً یک متغیر علامت‌دار توان باتری پیش‌بینی شود و سپس به دو بخش نامنفی شکسته شود. این کار قید را سخت‌تر از جریمه نرم اعمال می‌کند و در نسخه‌های بعدی توصیه می‌شود.

\section{باقیمانده توازن توان}
با جایگذاری خروجی در رابطه~\eqref{eq:power-balance}:
\begin{equation}
  r_{P}(t)=\hat{P}_{\mathrm{grid}}(t)+P_{\mathrm{PV}}(t)
  +\hat{P}_{\mathrm{dis}}(t)-\hat{P}_{\mathrm{ch}}(t)
  -P_{\mathrm{load}}(t).
  \label{eq:power-residual}
\end{equation}
هزینه متناظر
\begin{equation}
  \mathcal{L}_{P}=\frac{1}{N_f}\sum_{i=1}^{N_f}
  \left|r_P(t_i)\right|^2
  \label{eq:power-loss}
\end{equation}
است. نقاط $t_i$ می‌توانند همان نمونه‌های داده یا نقاط کمکی روی افق بهره‌برداری باشند. ایده نقاط هم‌مکان دقیقاً از چارچوب رئیسی و همکاران آمده است \pcite{raissi2019pinns}؛ این‌جا به‌جای شبکه مکانی-زمانی، شبکه زمانی بهره‌برداری داریم.

\section{باقیمانده وضعیت شارژ}
\begin{equation}
\begin{split}
  r_{\mathrm{SOC}}(t)=\widehat{\mathrm{SOC}}(t+1)-\mathrm{SOC}(t)
  &-\frac{\eta_{\mathrm{ch}}\hat{P}_{\mathrm{ch}}(t)\Delta t}{E_{\max}}\\
  &+\frac{\hat{P}_{\mathrm{dis}}(t)\Delta t}
  {\eta_{\mathrm{dis}}E_{\max}}.
\end{split}
\label{eq:soc-residual}
\end{equation}
میانگین مربع این مقدار، $\mathcal{L}_{\mathrm{SOC}}$ است. اگر $\mathrm{SOC}(t)$ خود خروجی مرحله قبل باشد، می‌توان مدل را روی یک افق چندگامی باز کرد. نسخه اول همین گزارش یک‌گامی است تا اشکال‌زدایی ساده‌تر بماند.

برای حدود، جریمه نرم
\begin{equation}
  \phi(z)=\max(0,z-z_{\max})^2+\max(0,z_{\min}-z)^2
  \label{eq:bound-penalty}
\end{equation}
به کار می‌رود. این قید سخت نیست. اگر نقض صفر لازم باشد، باید خروجی با سیگموید یا تصویر روی بازه مجاز تبدیل شود.

\section{مؤلفه اقتصادی}
یک هزینه اقتصادی ساده چنین است:
\begin{equation}
  \mathcal{L}_{\mathrm{economic}}
  =\frac{1}{N}\sum_{i=1}^{N}
  \Big(
    c_{\mathrm{buy}}(t_i)[\hat{P}_{\mathrm{grid}}(t_i)]_{+}
    -c_{\mathrm{sell}}(t_i)[-\hat{P}_{\mathrm{grid}}(t_i)]_{+}
    +\kappa(\hat{P}_{\mathrm{ch}}(t_i)+\hat{P}_{\mathrm{dis}}(t_i))
  \Big),
  \label{eq:economic-loss}
\end{equation}
که $[\,\cdot\,]_{+}$ قسمت مثبت و $\kappa$ جریمه تقریبی چرخه باتری است. این عبارت از روح مدل اقتصادی مقاله ریزشبکه می‌آید، اما همان تابع هدف چانه‌زنی چندسایت نیست \pcite{oladejo2019microgrid}. اگر چند مشارکت‌کننده وجود داشته باشد، پس از تولید برنامه کل می‌توان سود ائتلاف را با وزن‌های چانه‌زنی تقسیم کرد. آن مرحله بیرون از گرادیان شبکه نگه داشته می‌شود تا آموزش ناپایدار نشود.

\section{تابع هزینه نهایی}
\begin{equation}
\begin{split}
  \mathcal{L}={}&
  \lambda_{d}\mathcal{L}_{\mathrm{data}}
  +\lambda_{P}\mathcal{L}_{P}
  +\lambda_{S}\mathcal{L}_{\mathrm{SOC}}\\
  &+\lambda_{c}\mathcal{L}_{\mathrm{constraint}}
  +\lambda_{e}\mathcal{L}_{\mathrm{economic}}.
\end{split}
\label{eq:total-loss}
\end{equation}
پیشنهاد شروع این است که همه ورودی و خروجی‌ها نرمال شوند و وزن‌ها از مرتبه یک آغاز گردند. سپس با مجموعه اعتبارسنجی، $\lambda_{P}$ و $\lambda_{S}$ افزایش داده شوند تا باقیمانده فیزیک پایین بیاید، بی‌آن‌که خطای داده فروپاشد. رئیسی و همکاران نیز نشان داده‌اند که طراحی هزینه و نمونه‌برداری نقاط فیزیکی از ارکان موفقیت است \pcite{raissi2019pinns}.

\section{مراحل آموزش}
\begin{enumerate}
  \item گردآوری یا شبیه‌سازی بار، تولید، قیمت و باتری؛
  \item یکسان‌سازی واحد، علامت و گام زمانی؛
  \item تقسیم زمانی به آموزش، اعتبارسنجی و آزمون؛
  \item نرمال‌سازی فقط با آمار مجموعه آموزش؛
  \item مقداردهی شبکه و وزن‌های هزینه؛
  \item گذر رو به جلو، محاسبه همه مؤلفه‌ها، انتشار معکوس؛
  \item ثبت همزمان \lr{MSE}، $E_P$ و نرخ نقض قید؛
  \item توقف بر اساس اعتبارسنجی، نه فقط آموزش.
\end{enumerate}
