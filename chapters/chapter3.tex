\section{ورودی و خروجی}
در آزمایش عددی، به‌جای خود زمان، ساعت شبانه‌روز به بازه صفر تا یک برده شده است: $t_{\mathrm{h}}=(t \bmod 24)/24$. بردار ورودی چنین است:
\begin{equation}
  \bm{x}_t=
  \left[t_{\mathrm{h}},\,P_{\mathrm{load}}(t),\,P_{\mathrm{PV}}(t),\,
  c_{\mathrm{buy}}(t),\,c_{\mathrm{sell}}(t),\,\mathrm{SOC}(t)\right].
  \label{eq:model-input}
\end{equation}
خروجی شبکه چهار عدد است:
\begin{equation}
  \hat{\bm{y}}_t=
  \left[\hat{P}_{\mathrm{grid}}(t),\,
  \hat{P}_{\mathrm{ch}}(t),\,
  \hat{P}_{\mathrm{dis}}(t),\,
  \widehat{\mathrm{SOC}}(t+1)\right].
  \label{eq:model-output}
\end{equation}
اگر برای بعضی خروجی‌ها نمونه مرجع نباشد، می‌توان آن بخش را از خطای داده برداشت و فقط با قانون فیزیک محدود کرد.

برای جلوگیری از شارژ و تخلیه هم‌زمان می‌توان به‌جای دو توان جدا، یک توان علامت‌دار برای باتری پیش‌بینی کرد و بعد آن را به دو بخش نامنفی شکست. این کار در آزمایش فعلی انجام نشده است.

\section{خطای توازن توان}
با گذاشتن خروجی شبکه در رابطه~\eqref{eq:power-balance} اختلاف توازن چنین تعریف می‌شود:
\begin{equation}
  r_{P}(t)=\hat{P}_{\mathrm{grid}}(t)+P_{\mathrm{PV}}(t)
  +\hat{P}_{\mathrm{dis}}(t)-\hat{P}_{\mathrm{ch}}(t)
  -P_{\mathrm{load}}(t).
  \label{eq:power-residual}
\end{equation}
هزینه متناظر میانگین مربع همین اختلاف است:
\begin{equation}
  \mathcal{L}_{P}=\frac{1}{N_f}\sum_{i=1}^{N_f}
  \left|r_P(t_i)\right|^2.
  \label{eq:power-loss}
\end{equation}
ساعت‌های $t_i$ در این آزمایش همان نمونه‌های آموزش‌اند. فکر استفاده از نقاط کمکی از کار رئیسی و همکاران آمده است \pcite{raissi2019pinns}؛ اینجا به‌جای شبکه مکانی، فقط محور زمان بهره‌برداری وجود دارد.

\section{خطای شارژ باتری}
\begin{equation}
\begin{split}
  r_{\mathrm{SOC}}(t)=\widehat{\mathrm{SOC}}(t+1)-\mathrm{SOC}(t)
  &-\frac{\eta_{\mathrm{ch}}\hat{P}_{\mathrm{ch}}(t)\Delta t}{E_{\max}}\\
  &+\frac{\hat{P}_{\mathrm{dis}}(t)\Delta t}
  {\eta_{\mathrm{dis}}E_{\max}}.
\end{split}
\label{eq:soc-residual}
\end{equation}
میانگین مربع این مقدار، $\mathcal{L}_{\mathrm{SOC}}$ نامیده می‌شود. مدل فعلی یک‌ساعته است؛ یعنی شارژ ساعت بعد فقط از روی همان ساعت ساخته می‌شود، نه از زنجیره چندساعته.

برای سقف و کف شارژ از جریمه نرم استفاده می‌شود:
\begin{equation}
  \phi(z)=\max(0,z-z_{\max})^2+\max(0,z_{\min}-z)^2.
  \label{eq:bound-penalty}
\end{equation}
جریمه نرم یعنی خروج از بازه ممنوع قطعی نیست، فقط هزینه آموزش را بالا می‌برد. به همین دلیل در آزمایش بعدی دیده می‌شود که درصد نقض قید صفر نشده است. اگر رعایت سخت لازم باشد باید خروجی را با تابعی مانند سیگموید به داخل بازه برد.

\section{هزینه اقتصادی}
یک هزینه ساده بهره‌برداری چنین نوشته می‌شود:
\begin{equation}
  \mathcal{L}_{\mathrm{economic}}
  =\frac{1}{N}\sum_{i=1}^{N}
  \Big(
    c_{\mathrm{buy}}(t_i)[\hat{P}_{\mathrm{grid}}(t_i)]_{+}
    -c_{\mathrm{sell}}(t_i)[-\hat{P}_{\mathrm{grid}}(t_i)]_{+}
    +\kappa(\hat{P}_{\mathrm{ch}}(t_i)+\hat{P}_{\mathrm{dis}}(t_i))
  \Big).
  \label{eq:economic-loss}
\end{equation}
علامت $[\,\cdot\,]_{+}$ قسمت مثبت است و $\kappa$ جریمه تقریبی رفت‌وبرگشت انرژی در باتری است. این رابطه از جنس هزینه خرید و فروش است، نه همان تابع چانه‌زنی چندسایت مقاله اولادجو و فولی \pcite{oladejo2019microgrid}. در آزمایش فصل بعد این عبارت فقط بعد از آموزش به‌عنوان معیار گزارش شده و داخل مشتق وزن‌ها نبوده است.

\section{تابع هزینه کامل}
شکل کلی هزینه چنین است:
\begin{equation}
\begin{split}
  \mathcal{L}={}&
  \lambda_{d}\mathcal{L}_{\mathrm{data}}
  +\lambda_{P}\mathcal{L}_{P}
  +\lambda_{S}\mathcal{L}_{\mathrm{SOC}}\\
  &+\lambda_{c}\mathcal{L}_{\mathrm{constraint}}
  +\lambda_{e}\mathcal{L}_{\mathrm{economic}}.
\end{split}
\label{eq:total-loss}
\end{equation}
در اجرای گزارش‌شده $\lambda_{e}$ صفر بوده است. وزن بخش فیزیک پیش از اجرا ثابت گذاشته شد و روی مجموعه اعتبارسنجی تنظیم نشد.

\section{مراحل کار در آزمایش}
\begin{enumerate}
  \item ساخت بیست‌ویک شبانه‌روز داده ساعتی؛
  \item یکسان کردن واحد و علامت توان‌ها؛
  \item تقسیم زمانی به آموزش، نگه‌داشته و آزمون؛
  \item افزودن نویز فقط روی برچسب آموزش؛
  \item آموزش دو شبکه هم‌شکل، یکی بدون فیزیک و یکی با فیزیک؛
  \item ارزیابی روی ساعات آزمون با چهار معیار بالا.
\end{enumerate}
نرمال‌سازی کامل ورودی‌ها و توقف زودهنگام بر اساس مجموعه نگه‌داشته در این اجرا انجام نشد.
