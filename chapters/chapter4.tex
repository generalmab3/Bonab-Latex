\section{نمونه معادل در پای‌تورچ}
کد~\ref{lst:pytorch-model} اجرا نشده است. عرض لایه در آن ۶۴ است، در حالی که شبکه آزمایش ۴۸ نورون دارد.

\begin{latin}
\begin{lstlisting}[caption={\protect\lr{Optional PyTorch sketch; not used for the reported table}},label={lst:pytorch-model}]
import torch
from torch import nn


class EnergyPINN(nn.Module):
    def __init__(self, n_in=6, n_out=4, width=64, depth=4):
        super().__init__()
        layers = [nn.Linear(n_in, width), nn.Tanh()]
        for _ in range(depth - 1):
            layers += [nn.Linear(width, width), nn.Tanh()]
        layers += [nn.Linear(width, n_out)]
        self.network = nn.Sequential(*layers)

    def forward(self, x):
        return self.network(x)


def upper_lower_penalty(z, lower, upper):
    above = torch.relu(z - upper).pow(2)
    below = torch.relu(lower - z).pow(2)
    return (above + below).mean()
\end{lstlisting}
\end{latin}

کد~\ref{lst:physics-loss} همان رابطه‌های فصل قبل را با نماد تنسوری می‌نویسد.

\begin{latin}
\begin{lstlisting}[caption={\protect\lr{Power and battery residuals}},label={lst:physics-loss}]
def physics_losses(model, batch, dt, e_max, eta_ch, eta_dis):
    prediction = model(batch["x"])
    p_grid, p_ch, p_dis, soc_next = prediction.split(1, dim=1)

    power_residual = (
        p_grid + batch["p_pv"] + p_dis
        - p_ch - batch["p_load"]
    )

    expected_soc = (
        batch["soc"]
        + eta_ch * p_ch * dt / e_max
        - p_dis * dt / (eta_dis * e_max)
    )
    soc_residual = soc_next - expected_soc

    loss_power = power_residual.pow(2).mean()
    loss_soc = soc_residual.pow(2).mean()
    loss_bounds = upper_lower_penalty(
        soc_next, batch["soc_min"], batch["soc_max"]
    )
    return prediction, loss_power, loss_soc, loss_bounds
\end{lstlisting}
\end{latin}

\section{چگونه داده ساخته شد}
شبیه‌ساز بیست‌ویک شبانه‌روز را ساعت‌به‌ساعت می‌سازد. بار در طول روز بالا و پایین می‌رود و عصر کمی بیشتر است. تولید خورشیدی فقط در روز ظاهر می‌شود. قیمت خرید در ساعات اوج گران‌تر است. برنامه مرجع با یک قاعده ساده به دست می‌آید: اگر خورشید از مصرف بیشتر باشد باتری تا سقف شارژ می‌شود؛ اگر کمتر باشد از باتری برداشت می‌شود؛ باقی‌مانده از شبکه می‌آید. ظرفیت باتری بیست کیلووات‌ساعت، بازده شارژ و تخلیه $0.95$ و بازه شارژ از $0.20$ تا $0.90$ است.

روی برچسب‌های مجموعه آموزش نویز گاوسی با انحراف معیار $0.35$ گذاشته شد تا بخش فیزیک چیزی برای اصلاح داشته باشد. مجموعه آزمون این نویز را ندارد. از ۵۰۴ ساعت کل، ۶۰ درصد آموزش، ۲۰ درصد کنارگذاشته و ۲۰ درصد آزمون است؛ یعنی ۱۰۲ ساعت آزمون. ساعت‌های مجاور به‌هم ریخته نشده‌اند. تولید خورشیدی منفی بریده شده است.

ورودی‌ها استانداردسازی کامل نشدند. فقط ساعت شبانه‌روز به بازه صفر تا یک برده شد. مجموعه کنارگذاشته برای تنظیم وزن به کار نرفت و وزن فیزیک از پیش ثابت ماند.

\section{دو مدلی که مقایسه شدند}
هر دو شبکه شکل یکسان داشتند:
\begin{enumerate}
  \item شبکه داده‌محور که فقط خطای برچسب را می‌بیند؛
  \item شبکه فیزیک‌آگاه که علاوه بر آن، خطای توازن توان، خطای دینامیک باتری و جریمه خروج از بازه شارژ را هم می‌بیند.
\end{enumerate}
حل‌کننده بهینه‌سازی کلاسیک به‌عنوان مرجع سوم اجرا نشد. هزینه اقتصادی رابطه~\eqref{eq:economic-loss} فقط بعد از آموزش حساب شد.

\section{معیارها}
میانگین مربع اختلاف خروجی با مرجع:
\begin{equation}
  \mathrm{MSE}=\frac{1}{N}\sum_{i=1}^{N}
  \left\|\hat{\bm{y}}_i-\bm{y}_i\right\|_2^2.
  \label{eq:mse}
\end{equation}
میانگین قدرمطلق اختلاف توازن توان:
\begin{equation}
  E_{P}=\frac{1}{N}\sum_{i=1}^{N}|r_P(t_i)|.
  \label{eq:physics-metric}
\end{equation}
درصد ساعت‌هایی که شارژ پیش‌بینی‌شده از بازه مجاز بیرون است:
\begin{equation}
  R_{v}=\frac{N_{\mathrm{violated}}}{N}\times 100.
  \label{eq:violation-rate}
\end{equation}
میانگین هزینه بهره‌برداری نیز آمده است. فقط یک بذر گزارش شده و پراکندگی چند اجرا در دست نیست.

\section{نتیجه عددی}
دو مدل با تقسیم یکسان آموزش دیدند و روی آزمون ندیده سنجیده شدند. تنظیم وزن، تحلیل حساسیت و آزمایش کمبود داده انجام نشد.

جدول~\ref{tab:real-results} نشان می‌دهد مدل فیزیک‌آگاه میانگین مربع خطا را از $0.0099$ به $0.0167$ رسانده، یعنی برازش برچسب کمی بدتر شده است. در عوض میانگین قدرمطلق خطای توازن توان از $0.082$ به $0.033$ کیلووات و درصد خروج شارژ از بازه از $47.1$ به $21.6$ رسیده است. هزینه متوسط هم اندکی کمتر شده است. این همان بده‌بستانی است که از گذاشتن فیزیک روی برچسب نویزی انتظار می‌رود.
